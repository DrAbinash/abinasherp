   143→// ── Second-monitor auto-placement ──────────────────────────────────────────
   144→// Opens `url` positioned on the second monitor automatically, with no manual
   145→// dragging required, using the best method the browser/connection allows:
   146→//
   147→//  1. Window Management API (window.getScreenDetails) — exact placement on
   148→//     whichever screen isn't currently showing this window, with automatic
   149→//     fullscreen. Only available in a SECURE context (https://) in Chrome/Edge
   150→//     100+, and only after a one-time browser permission prompt. This ERP
   151→//     currently runs on plain http:// on the LAN, so this path is unavailable
   152→//     today — it activates automatically the moment the clinic moves to
   153→//     HTTPS (e.g. via Tailscale HTTPS or a proper certificate), no code
   154→//     changes needed then.
   155→//  2. Practical fallback (works today, over plain HTTP, no permissions):
   156→//     assumes the second monitor is extended to the right of the primary —
   157→//     the default Windows/most-common clinic PC setup — and opens the window
   158→//     at that offset, sized to fill a typical monitor. Not pixel-perfect on
   159→//     unusual monitor arrangements, but lands correctly for the common case
   160→//     with zero dragging. A "Fullscreen" button on the display page itself
   161→//     (one click, no drag) covers the rest.
   162→async function openOnSecondMonitor(url: string, windowName: string): Promise<void> {
   163→  const wm = (window as unknown as { getScreenDetails?: () => Promise<{
   164→    screens: Array<{ availLeft: number; availTop: number; availWidth: number; availHeight: number }>;
   165→    currentScreen: { availLeft: number; availTop: number };
   166→  }> }).getScreenDetails;
   167→
   168→  if (wm && window.isSecureContext) {
   169→    try {
   170→      const details = await wm();
   171→      const other = details.screens.find(
   172→        (s) => s.availLeft !== details.currentScreen.availLeft || s.availTop !== details.currentScreen.availTop
   173→      ) ?? details.screens[0];
   174→      const win = window.open(
   175→        url,
   176→        windowName,
   177→        `left=${other.availLeft},top=${other.availTop},width=${other.availWidth},height=${other.availHeight}`
   178→      );
   179→      if (win) {
   180→        win.addEventListener("load", () => {
   181→          try { win.document.documentElement.requestFullscreen?.(); } catch { /* best effort */ }
   182→        });
   183→        return;
   184→      }
   185→    } catch {
   186→      // Permission denied or API unavailable this call — fall through to heuristic.
   187→    }
   188→  }
   189→
   190→  // Fallback: place just past the primary screen's right edge.
   191→  const left = window.screen.width;
   192→  window.open(url, windowName, `left=${left},top=0,width=1024,height=900`);
   193→}

   198→function useDebounce<T>(value: T, delay = 300) {
   199→  const [debounced, setDebounced] = useState(value);
   200→  useEffect(() => {
   201→    const t = setTimeout(() => setDebounced(value), delay);
   202→    return () => clearTimeout(t);
   203→  }, [value, delay]);
   204→  return debounced;
   205→}

   258→function escapeHtml(s: string) {
   259→  return String(s).replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]!));

   261→function calcAge(dateOfBirth: string, ageValue?: number | null, ageUnit?: string | null): string {
   262→  if (ageValue != null && ageUnit) {
   263→    if (ageUnit === "years") return ageValue > 0 ? `${ageValue} Yrs` : "";
   264→    if (ageUnit === "months") return `${ageValue} Mo`;
   265→    if (ageUnit === "days") return `${ageValue} D`;
   266→  }
   267→  if (!dateOfBirth) return "";
   268→  const dob = new Date(dateOfBirth);
   269→  if (isNaN(dob.getTime())) return "";
   270→  const now = new Date();
   271→  let y = now.getFullYear() - dob.getFullYear();
   272→  const m = now.getMonth() - dob.getMonth();
   273→  if (m < 0 || (m === 0 && now.getDate() < dob.getDate())) y--;
   274→  return y > 0 ? `${y} Yrs` : "";
   275→}

   283→function code128SVG(value: string): string {
   284→  // Lightweight inline barcode renderer (Code 128 B subset, digits + uppercase + symbols).
   285→  // For production accuracy use a library; this gives a scan-friendly visual barcode.
   286→  const PATTERNS: string[] = ["11011001100","11001101100","11001100110","10010011000","10010001100","10001001100","10011001000","10011000100","10001100100","11001001000","11001000100","11000100100","10110011100","10011011100","10011001110","10111001100","10011101100","10011100110","11001110010","11001011100","11001001110","11011100100","11001110100","11101101110","11101001100","11100101100","11100100110","11101100100","11100110100","11100110010","11011011000","11011000110","11000110110","10100011000","10001011000","10001000110","10110001000","10001101000","10001100010","11010001000","11000101000","11000100010","10110111000","10110001110","10001101110","10111011000","10111000110","10001110110","11101110110","11010001110","11000101110","11011101000","11011100010","11011101110","11101011000","11101000110","11100010110","11101101000","11101100010","11100011010","11101111010","11001000010","11110001010","10100110000","10100001100","10010110000","10010000110","10000101100","10000100110","10110010000","10110000100","10011010000","10011000010","10000110100","10000110010","11000010010","11001010000","11110111010","11000010100","10001111010","10100111100","10010111100","10010011110","10111100100","10011110100","10011110010","11110100100","11110010100","11110010010","11011011110","11011110110","11110110110","10101111000","10100011110","10001011110","10111101000","10111100010","11110101000","11110100010","10111011110","10111101110","11101011110","11110101110","11010000100","11010010000","11010011100","1100011101011"];
   287→  // 0..127 chars: ASCII offset 32 → index 0
   288→  const data: number[] = [];
   289→  for (const ch of value) data.push(Math.max(0, Math.min(94, ch.charCodeAt(0) - 32)));
   290→  let checksum = 104; // START B
   291→  data.forEach((c, i) => { checksum += c * (i + 1); });
   292→  checksum = checksum % 103;
   293→  const codes = [104, ...data, checksum, 106]; // START B + data + checksum + STOP
   294→  let bars = "";
   295→  let x = 0;
   296→  const W = 1.6, H = 50;
   297→  for (const code of codes) {
   298→    const pat = PATTERNS[code];
   299→    for (let i = 0; i < pat.length; i++) {
   300→      if (pat[i] === "1") bars += `<rect x="${x.toFixed(2)}" y="0" width="${W}" height="${H}" fill="#000"/>`;
   301→      x += W;
   302→    }
   303→  }
   304→  const totalW = x;
   305→  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${totalW} ${H}" width="${totalW}" height="${H}">${bars}</svg>`;
   306→}

   333→// Render the per-department queue token slip on the configured Token Printer.
   334→// One combined slip lists every department the patient needs to visit, with
   335→// the token number and the room/counter for each. Falls back to the legacy
   336→// single-token layout when only `tokenNo` is present (back-compat for bills
   337→// generated before the per-test token rollout).
   338→async function printToken(b: LastBill, clinic: ClinicLite) {
   339→  const p = await getPrinterSettings();
   340→  if (p.tokenEnabled === "false") return;
   341→  // Group testTokens by (department, roomNumber, tokenNo) — we don't need to
   342→  // print the same dept token twice when a bill has two pathology tests that
   343→  // share one department-level token... wait, current generator emits one
   344→  // token PER orderTest. Until the per-department mode lands we de-dupe by
   345→  // department here so the slip stays compact.
   346→  const seen = new Set<string>();
   347→  const dedupedTokens = (b.testTokens ?? []).filter((t) => {
   348→    const key = `${t.department}::${t.tokenNo}`;
   349→    if (seen.has(key)) return false;
   350→    seen.add(key);
   351→    return true;
   352→  });
   353→
   354→  // Nothing to print
   355→  if (dedupedTokens.length === 0 && b.tokenNo == null) return;
   356→
   357→  const ageStr = calcAge(b.patient.dateOfBirth, b.patient.ageValue, b.patient.ageUnit);
   358→  const ageGender = [ageStr, b.patient.gender].filter(Boolean).join(" / ").toUpperCase();
   359→  const headerLines = `
   360→    <div class="clinic">${escapeHtml(clinic?.name || "Diagnostic Centre")}</div>
   361→    <div class="patient"><strong>${escapeHtml(b.patient.firstName)} ${escapeHtml(b.patient.lastName)}</strong>${ageGender ? ` &middot; ${escapeHtml(ageGender)}` : ""}</div>
   362→    <div class="meta">ID: ${escapeHtml(b.patient.patientId)} &middot; Bill: ${escapeHtml(String(b.billNumber).replace(/^BILL-?/i, "").replace(/-/g, ""))}</div>
   363→    ${b.doctorName ? `<div class="meta">Ref: Dr. ${escapeHtml(b.doctorName)}</div>` : ""}
   364→    <div class="meta">${new Date().toLocaleString("en-IN")}</div>`;
   365→
   366→  let body = "";
   367→  if (dedupedTokens.length > 0) {
   368→    // Combined per-department slip
   369→    const rows = dedupedTokens.map((t) => `
   370→      <tr>
   371→        <td class="dept">${escapeHtml(t.department)}</td>
   372→        <td class="num">#${String(t.tokenNo).padStart(3, "0")}</td>
   373→        <td class="room">${t.roomNumber ? `Room ${escapeHtml(t.roomNumber)}${t.floorLabel ? `<br><span style="font-size:9px;color:#666;font-weight:400">${escapeHtml(t.floorLabel)}</span>` : ""}` : "—"}</td>
   374→      </tr>`).join("");
   375→    body = `
   376→      <div class="title">QUEUE TOKENS</div>
   377→      <table class="tokens"><tbody>${rows}</tbody></table>
   378→      <div class="hint">Please proceed to the indicated room and wait for your number to be called.</div>`;
   379→  } else if (b.tokenNo != null) {
   380→    // Legacy single-number fallback
   381→    body = `
   382→      <div class="title">QUEUE TOKEN</div>
   383→      <div class="bignum">${String(b.tokenNo).padStart(3, "0")}</div>
   384→      <div class="hint">Please wait for your token to be called.</div>`;
   385→  }
   386→
   387→  const html = `<!doctype html><html><head><meta charset="utf-8"><title>Token Slip — ${escapeHtml(b.billNumber)}</title>
   388→    <style>
   389→      @page { size: 80mm auto; margin: 4mm; }
   390→      body { font-family: Arial, sans-serif; margin:0; padding:6px; text-align:center; color:#000; font-size:11px; }
   391→      .clinic { font-size:13px; font-weight:800; border-bottom:1px dashed #000; padding-bottom:4px; margin-bottom:5px; }
   392→      .patient { font-size:12px; margin:1px 0; }
   393→      .meta { font-size:10px; color:#444; margin:1px 0; }
   394→      .title { margin-top:8px; font-size:10px; letter-spacing:2px; color:#444; }
   395→      .bignum { font-size:64px; font-weight:900; line-height:1; margin:4px 0; }
   396→      table.tokens { width:100%; border-collapse:collapse; margin-top:6px; }
   397→      table.tokens td { padding:4px 2px; border-bottom:1px dashed #ccc; vertical-align:middle; }
   398→      table.tokens td.dept { text-align:left; font-weight:700; font-size:11px; text-transform:uppercase; }
   399→      table.tokens td.num  { text-align:center; font-weight:900; font-size:18px; letter-spacing:1px; white-space:nowrap; }
   400→      table.tokens td.room { text-align:right; font-size:10px; color:#1e40af; font-weight:700; white-space:nowrap; }
   401→      .hint { margin-top:6px; padding-top:4px; border-top:1px dashed #000; font-size:9px; color:#555; }
   402→    </style></head><body>
   403→    ${headerLines}
   404→    ${body}
   405→  </body></html>`;
   406→  const w = window.open("", "_blank", printerWindowFeatures(p.tokenPrinter));
   407→  if (!w) return openPrintWindow(html);
   408→  w.document.open(); w.document.write(html); w.document.close(); w.onload = () => { w.focus(); w.print(); setTimeout(() => w.close(), 400); };
   409→}

  1431→  function addPackage(pkg: Pkg) {
  1432→    // Package effective price = MRP - %% - flat ₹.
  1433→    const afterPct = pkg.price - (pkg.price * (pkg.discountPct ?? 0)) / 100;
  1434→    const effective = Math.max(0, afterPct - (pkg.discountAmount ?? 0));
  1435→    const count = pkg.tests.length || 1;
  1436→
  1437→    // If any test inside this package carries its own discount override, honour
  1438→    // that on a per-line basis. Otherwise fall back to the historical even
  1439→    // split of the package's effective price.
  1440→    const anyOverride = pkg.tests.some(
  1441→      (t) => Number(t.discountPct ?? 0) > 0 || Number(t.discountAmount ?? 0) > 0,
  1442→    );
  1443→    const computeLinePrice = (t: PkgTest) => {
  1444→      if (anyOverride) {
  1445→        const base = Number(t.price);
  1446→        const after = base - (base * Number(t.discountPct ?? 0)) / 100 - Number(t.discountAmount ?? 0);
  1447→        return Math.max(0, after);
  1448→      }
  1449→      return effective / count;
  1450→    };
  1451→
  1452→    const existingIds = new Set(selectedTests.map((s) => s.testId));
  1453→    const toAdd: SelectedTest[] = pkg.tests
  1454→      .filter((t) => !existingIds.has(t.id))
  1455→      .map((t) => ({ testId: t.id, name: t.name, code: t.code, price: computeLinePrice(t), category: t.category, source: "package" as const }));
  1456→    if (toAdd.length === 0) {
  1457→      toast({ title: "All tests in this package already added" });
  1458→      return;
  1459→    }
  1460→    setSelectedTests((prev) => [...prev, ...toAdd]);
  1461→    setSelectedPackages((prev) => [...prev, { packageId: pkg.id, name: pkg.name, testIds: toAdd.map((t) => t.testId) }]);
  1462→    toast({ title: `Package "${pkg.name}" added (${toAdd.length} tests)` });
  1463→  }
