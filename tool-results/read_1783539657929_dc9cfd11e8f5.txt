    10→/**
    11→ * Reject super-admin bill mutations if the USB pen-drive gate is enforced and
    12→ * the request does not carry a valid X-SA-USB-Key header. Returns true when
    13→ * the response has been sent (caller should return). Centralised here so the
    14→ * super-admin token flow used by /:id/super-edit and DELETE /:id matches the
    15→ * rest of the super-admin surface.
    16→ *
    17→ * If the user has remoteLoginEnabled, the USB check is skipped on routes that
    18→ * already validated the session token via requireSuperAdmin. For the direct
    19→ * bill-mutation routes the caller should just pass the USB key header.
    20→ */
    21→function rejectIfUsbMissing(req: Request, res: Response): boolean {
    22→  if (!isUsbGateEnforced()) return false;
    23→  const usb = getUsbKeyHeader(req);
    24→  if (!usb || !isValidUsbKey(usb)) {
    25→    res.status(401).json({ error: "USB key required" });
    26→    return true;
    27→  }
    28→  return false;
    29→}

    76→async function resolveLedgerForOrder(orderId: number): Promise<number> {
    77→  const [o] = await db.select().from(ordersTable).where(eq(ordersTable.id, orderId));
    78→  if (o?.ledgerId) return o.ledgerId;

    84→  return getWalkInLedgerId();

    87→/**
    88→ * Bill numbers are pure-numeric: YYYYMM + 4-digit sequence (e.g. 2026050001).
    89→ * Older rows in the DB may still carry the legacy `BILL-YYYYMM-####` prefix;
    90→ * `parseBillNumberParts` handles both shapes so the renumber logic keeps
    91→ * working across the migration.
    92→ */
    93→export async function generateBillNumber(_ledgerId: number): Promise<string> {
    94→  const date = new Date();
    95→  const yyyymm = `${date.getFullYear()}${String(date.getMonth() + 1).padStart(2, "0")}`;
    96→  // Use the global MAX across ALL numeric bills, not a per-ledger count.
    97→  // COUNT per ledger breaks when multiple ledgers share the same bill_number
    98→  // unique space — ledger A and B independently arrive at the same sequence
    99→  // number and collide on the UNIQUE constraint.
   100→  const [row] = await db
   101→    .select({ maxBill: sql<string | null>`MAX(bill_number)` })
   102→    .from(billsTable)
   103→    .where(sql`bill_number ~ '^[0-9]+$'`);
   104→  let seq = 1;
   105→  if (row?.maxBill) {
   106→    const parts = parseBillNumberParts(row.maxBill);
   107→    if (parts) seq = parts.seq + 1;
   108→  }
   109→  return `${yyyymm}${String(seq).padStart(4, "0")}`;
   110→}
   111→
   112→function parseBillNumberParts(billNumber: string): { monthPrefix: string; seq: number } | null {
   113→  const legacy = billNumber.match(/^BILL-(\d{6})-(\d+)$/);
   114→  if (legacy) return { monthPrefix: legacy[1]!, seq: Number(legacy[2]) };
   115→  const numeric = billNumber.match(/^(\d{6})(\d{4,})$/);
   116→  if (numeric) return { monthPrefix: numeric[1]!, seq: Number(numeric[2]) };
   117→  return null;
   118→}

   120→async function buildBill(bill: typeof billsTable.$inferSelect) {
   121→  const [patient] = await db.select().from(patientsTable).where(eq(patientsTable.id, bill.patientId));
   122→  const [order] = await db.select().from(ordersTable).where(eq(ordersTable.id, bill.orderId));
   123→
   124→  const payments = await db.select().from(paymentsTable).where(eq(paymentsTable.billId, bill.id)).orderBy(desc(paymentsTable.createdAt));
   125→
   126→  let orderDetails = null;
   127→  if (order) {
   128→    const orderTestRows = await db
   129→      .select({ orderTest: orderTestsTable, test: testsTable })
   130→      .from(orderTestsTable)
   131→      .leftJoin(testsTable, eq(orderTestsTable.testId, testsTable.id))
   132→      .where(eq(orderTestsTable.orderId, order.id));
   133→
   134→    let doctor = null;
   135→    if (order.doctorId) {
   136→      const [d] = await db.select().from(doctorsTable).where(eq(doctorsTable.id, order.doctorId));
   137→      doctor = d ?? null;
   138→    }
   139→
   140→    orderDetails = {
   141→      ...order,
   142→      totalAmount: Number(order.totalAmount),
   143→      patient: patient ? sanitizePatient(patient) : null,
   144→      doctor,
   145→      tests: orderTestRows.map((ot) => ({
   146→        ...ot.orderTest,
   147→        price: Number(ot.orderTest.price),
   148→        displayName: ot.orderTest.displayName ?? null,
   149→        test: ot.test ? { ...ot.test, price: Number(ot.test.price) } : null,
   150→      })),
   151→    };
   152→  }
   153→
   154→  return {
   155→    ...bill,
   156→    subtotal: Number(bill.subtotal),
   157→    discount: Number(bill.discount),
   158→    taxAmount: Number(bill.taxAmount),
   159→    totalAmount: Number(bill.totalAmount),
   160→    originalTotal: Number(bill.originalTotal),
   161→    paidAmount: Number(bill.paidAmount),
   162→    balanceAmount: Number(bill.balanceAmount),
   163→    patient: patient ? sanitizePatient(patient) : null,
   164→    order: orderDetails,
   165→    payments: payments.map((p) => ({ ...p, amount: Number(p.amount) })),
   166→  };
   167→}

   267→billsRouter.get("/", async (req, res) => {
   268→  const parsed = ListBillsQueryParams.safeParse(req.query);
   269→  if (!parsed.success) {
   270→    res.status(400).json({ error: "Invalid query params" });
   271→    return;
   272→  }
   273→  const { status, patientId, page = 1, limit = 20 } = parsed.data;
   274→  const offset = (page - 1) * limit;
   275→
   276→  // Extra filters not in the strict zod schema: dues-only + date range. These
   277→  // are read directly from req.query so we don't have to round-trip OpenAPI
   278→  // codegen. Powers the Dues report page.
   279→  const dueOnly = req.query.dueOnly === "1" || req.query.dueOnly === "true";
   280→  // When no specific status filter is set, callers can pass excludeCancelled=1
   281→  // to suppress cancelled rows (default view on the Bills page).
   282→  const excludeCancelled = req.query.excludeCancelled === "1" || req.query.excludeCancelled === "true";
   283→  const dateFrom = typeof req.query.dateFrom === "string" && req.query.dateFrom ? req.query.dateFrom : null;
   284→  const dateTo = typeof req.query.dateTo === "string" && req.query.dateTo ? req.query.dateTo : null;
   285→  const dateField = req.query.dateField === "due" ? "due" : "created";
   286→  const createdBy = typeof req.query.createdBy === "string" && req.query.createdBy ? req.query.createdBy : null;
   287→
   288→  // ISO date strings only — basic guard against SQL injection via the raw fragments.
   289→  const isoDateRe = /^\d{4}-\d{2}-\d{2}$/;
   290→  if (dateFrom && !isoDateRe.test(dateFrom)) {
   291→    res.status(400).json({ error: "dateFrom must be YYYY-MM-DD" });
   292→    return;
   293→  }
   294→  if (dateTo && !isoDateRe.test(dateTo)) {
   295→    res.status(400).json({ error: "dateTo must be YYYY-MM-DD" });
   296→    return;
   297→  }
   298→
   299→  const conditions: (ReturnType<typeof eq> | ReturnType<typeof gt> | ReturnType<typeof sql>)[] = [];
   300→  if (status) conditions.push(eq(billsTable.status, status));
   301→  else if (excludeCancelled) conditions.push(ne(billsTable.status, "cancelled"));
   302→  if (patientId) conditions.push(eq(billsTable.patientId, patientId));
   303→  if (dueOnly) {
   304→    // Only include bills with an actual outstanding balance AND not cancelled.
   305→    // Cancelled bills should have balanceAmount=0 already (zeroed on cancel),
   306→    // but the status exclusion is a safety net for any legacy rows.
   307→    conditions.push(gt(sql`${billsTable.balanceAmount}::numeric`, sql`0`));
   308→    conditions.push(ne(billsTable.status, "cancelled"));
   309→  }
   310→  if (createdBy) {
   311→    conditions.push(sql`${billsTable.createdByName} = ${createdBy}`);
   312→  }
   313→  if (dateFrom || dateTo) {
   314→    if (dateField === "due") {
   315→      // Bills only have a due date when one was set; bills with NULL dueDate are excluded by this filter.
   316→      if (dateFrom) conditions.push(sql`${billsTable.dueDate} >= ${dateFrom}` as unknown as ReturnType<typeof eq>);
   317→      if (dateTo) conditions.push(sql`${billsTable.dueDate} <= ${dateTo}` as unknown as ReturnType<typeof eq>);
   318→    } else {
   319→      // Compare on the date portion of created_at, in the server's local timezone.
   320→      if (dateFrom) conditions.push(sql`(${billsTable.createdAt})::date >= ${dateFrom}::date` as unknown as ReturnType<typeof eq>);
   321→      if (dateTo) conditions.push(sql`(${billsTable.createdAt})::date <= ${dateTo}::date` as unknown as ReturnType<typeof eq>);
   322→    }
   323→  }
   324→
   325→  const where = conditions.length > 0 ? and(...conditions) : undefined;
   326→
   327→  const [bills, countResult, totalsResult] = await Promise.all([
   328→    db.select().from(billsTable).where(where).orderBy(desc(billsTable.createdAt)).limit(limit).offset(offset),
   329→    db.select({ count: sql<number>`count(*)` }).from(billsTable).where(where),
   330→    // Aggregate totals across the FULL filtered set (not just the current page) — used by the Dues card.
   331→    db.select({
   332→      totalAmount: sql<string>`COALESCE(SUM(${billsTable.totalAmount}::numeric), 0)`,
   333→      paidAmount: sql<string>`COALESCE(SUM(${billsTable.paidAmount}::numeric), 0)`,
   334→      balanceAmount: sql<string>`COALESCE(SUM(${billsTable.balanceAmount}::numeric), 0)`,
   335→      refundAmount: sql<string>`COALESCE(SUM(${billsTable.refundAmount}::numeric), 0)`,
   336→    }).from(billsTable).where(where),
   337→  ]);
   338→
   339→  const billsWithDetails = await Promise.all(bills.map(buildBill));
   340→  res.json({
   341→    bills: billsWithDetails,
   342→    total: Number(countResult[0]?.count ?? 0),
   343→    page,
   344→    limit,
   345→    totals: {
   346→      totalAmount: Number(totalsResult[0]?.totalAmount ?? 0),
   347→      paidAmount: Number(totalsResult[0]?.paidAmount ?? 0),
   348→      balanceAmount: Number(totalsResult[0]?.balanceAmount ?? 0),
   349→      refundAmount: Number(totalsResult[0]?.refundAmount ?? 0),
   350→    },
   351→  });
   352→});

   356→  const parsed = CreateBillBody.safeParse(payload);
   357→  if (!parsed.success) {

   364→  const discountReasonNote = typeof payload?.discountReasonNote === "string" ? payload.discountReasonNote.trim() || null : null;

   367→  // ── Idempotency check — clientRef deduplication ───────────────────────────

   369→  // created for it. If yes, return the existing bill immediately (HTTP 200).
   370→  // This makes POST /api/bills safe to retry after a network timeout — the

   374→      sql`SELECT id FROM bills WHERE client_ref = ${clientRef} AND status != 'cancelled' LIMIT 1`

   384→          id: existingBillRow.id,
   385→          billNumber: existingBillRow.billNumber,
   386→          _idempotent: true, // signals the frontend this is a replayed response
   387→        });
   388→        return;
   389→      }
   390→    }
   391→  }

   393→  if (discount > 0 && !discountReason) {

   395→    return;
   396→  }

   398→  // Optional DICOM MWL fields captured at billing desk and written into the
   399→  // auto-generated radiology study rows. Not part of the Zod schema so we read

   401→  const rawDicom = payload?.dicomFields;
   402→  const dicomFields: { studyDescription?: string; bodyPart?: string; scheduledStationAETitle?: string; referringDoctor?: string } | undefined =
   403→    rawDicom && typeof rawDicom === "object"
   404→      ? {
   405→          studyDescription: typeof rawDicom.studyDescription === "string" ? rawDicom.studyDescription.trim() || undefined : undefined,
   406→          bodyPart: typeof rawDicom.bodyPart === "string" ? rawDicom.bodyPart.trim() || undefined : undefined,
   407→          scheduledStationAETitle: typeof rawDicom.scheduledStationAETitle === "string" ? rawDicom.scheduledStationAETitle.trim() || undefined : undefined,
   408→          referringDoctor: typeof rawDicom.referringDoctor === "string" ? rawDicom.referringDoctor.trim() || undefined : undefined,
   409→        }
   410→      : undefined;
   411→
   412→  const [order] = await db.select().from(ordersTable).where(eq(ordersTable.id, orderId));
   413→  if (!order) {
   414→    res.status(404).json({ error: "Order not found" });
   415→    return;
   416→  }
   417→
   418→  // Guard against double-billing:
   419→  // 1) Same order — prevents re-billing an order that already has a bill.
   420→  // 2) Same patient within 10s — catches rapid double-clicks that create two
   421→  //    separate orders. The frontend also has a synchronous generatingRef
   422→  //    guard, but this backend fence defends against network retries,
   423→  //    multi-tab races, and keyboard+click races.
   424→  const [existingBill] = await db
   425→    .select({ id: billsTable.id, billNumber: billsTable.billNumber })
   426→    .from(billsTable)
   427→    .where(and(eq(billsTable.orderId, orderId), ne(billsTable.status, "cancelled")))
   428→    .limit(1);
   429→  if (existingBill) {
   430→    res.status(409).json({
   431→      error: `This order already has an active bill (${existingBill.billNumber}). Cancel it first before creating a new one.`,
   432→    });
   433→    return;
   434→  }
   435→
   436→  const tenSecondsAgo = new Date(Date.now() - 60_000); // 60s window (was 10s) — covers slow connections and timeout retries
   437→  const [existingRecent] = await db
   438→    .select({ id: billsTable.id, billNumber: billsTable.billNumber })
   439→    .from(billsTable)
   440→    .where(and(
   441→      eq(billsTable.patientId, order.patientId),
   442→      ne(billsTable.status, "cancelled"),
   443→      gt(billsTable.createdAt, tenSecondsAgo),
   444→    ))
   445→    .limit(1);
   446→  if (existingRecent) {
   447→    res.status(409).json({
   448→      error: `A bill for this patient was just created (${existingRecent.billNumber}). Please wait a few seconds before billing again.`,
   449→    });
   450→    return;
   451→  }
   452→
   453→  // Refuse to bill against an order that contains discontinued / removed tests
   454→  // (matches the same guard applied at order creation in orders.ts). Catches
