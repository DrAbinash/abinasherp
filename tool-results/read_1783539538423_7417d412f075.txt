   210→function PermissionGuard() {
   211→  const [location, navigate] = useLocation();
   212→  useEffect(() => {
   213→    const session = readStaffSession();
   214→    if (!session) {
   215→      navigate("/portal", { replace: true });
   216→      return;
   217→    }
   218→    // Owner Dashboard is admin/super_admin only — redirect others to My Daily Summary.
   219→    const normalizedRole = normalizeRole(session.user.role);
   220→    if ((location === "/dashboard" || location === "/diagnostics") && !FULL_ACCESS_ROLES.has(normalizedRole)) {
   221→      navigate("/my-daily-summary", { replace: true });
   222→      return;
   223→    }
   224→    // Super Admin Portal is super_admin only
   225→    if (location.startsWith("/super-admin-portal") && normalizedRole !== "super_admin") {
   226→      navigate("/", { replace: true });
   227→      return;
   228→    }
   229→    // Redirect root "/" to user's default start page if configured and allowed
   230→    if (location === "/" && session.user.defaultStartPage && session.user.defaultStartPage !== "/" && canAccess(session, session.user.defaultStartPage)) {
   231→      navigate(session.user.defaultStartPage, { replace: true });
   232→      return;
   233→    }
   234→    const matched = longestMatchingNavPath(location, ERP_NAV_ORDER);
   235→    const pathToCheck = matched ?? location;
   236→    if (!canAccess(session, pathToCheck)) {
   237→      let target = "/";
   238→      if (session.user.defaultStartPage && canAccess(session, session.user.defaultStartPage)) {
   239→        target = session.user.defaultStartPage;
   240→      } else {
   241→        target = firstPermissionedPath(session, ERP_NAV_ORDER) ?? firstAllowedPath(session, ERP_NAV_ORDER);
   242→      }
   243→      navigate(target, { replace: true });
   244→    }
   245→  }, [location, navigate]);
   246→  return null;
   247→}
