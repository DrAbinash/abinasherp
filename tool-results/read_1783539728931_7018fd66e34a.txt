   317→function openReportWindow(c: Closure, clinic: ClinicLite) {
   318→  const html = buildSummarySlipHtml(c, clinic);
   319→  const w = window.open("", "_blank", "width=900,height=800");
   320→  if (!w) {
   321→    alert("Pop-up blocked. Allow pop-ups to print the report.");
   322→    return null;
   323→  }
   324→  w.document.open(); w.document.write(html); w.document.close();
   325→  return w;
   326→}
   327→
   328→function autoPrintSlip(c: Closure, clinic: ClinicLite) {
   329→  const w = openReportWindow(c, clinic);
   330→  if (!w) return;
   331→  w.onload = () => { w.focus(); w.print(); setTimeout(() => w.close(), 800); };
   332→}
   333→
   334→function downloadReportPdf(c: Closure, clinic: ClinicLite) {
   335→  const w = openReportWindow(c, clinic);
   336→  if (!w) return;
   337→  w.onload = () => {
   338→    w.focus();
   339→    // Modern browsers offer "Save as PDF" in the print dialog
   340→    w.print();
   341→  };
   342→}
