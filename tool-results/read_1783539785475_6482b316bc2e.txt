   401→formFRouter.get("/pending", async (req, res) => {
   402→  try {
   403→    const dateRange = String(req.query.dateRange ?? "today").trim() as "today" | "yesterday" | "dayBefore" | "7days" | "all";
   404→
   405→    const [settings] = await db.select().from(clinicSettingsTable).limit(1);
   406→    const formFTestIds: number[] = JSON.parse(settings?.formFTestIds ?? "[]");
   407→
   408→    if (formFTestIds.length === 0) {
   409→      res.json([]);
   410→      return;
   411→    }
   412→
   413→    // Compute IST date bounds for the chosen range
   414→    const now = new Date();
   415→    const istToday = now.toLocaleDateString("en-CA", { timeZone: "Asia/Kolkata" });
   416→    let startDate: string | null = null;
   417→    let endDate: string | null = null; // exclusive upper bound
   418→
   419→    if (dateRange === "today") {
   420→      startDate = istToday + "T00:00:00+05:30";
   421→      endDate = istToday + "T23:59:59.999+05:30";
   422→    } else if (dateRange === "yesterday") {
   423→      const y = new Date(now); y.setDate(y.getDate() - 1);
   424→      const ys = y.toLocaleDateString("en-CA", { timeZone: "Asia/Kolkata" });
   425→      startDate = ys + "T00:00:00+05:30";
   426→      endDate = ys + "T23:59:59.999+05:30";
   427→    } else if (dateRange === "dayBefore") {
   428→      const db = new Date(now); db.setDate(db.getDate() - 2);
   429→      const dbs = db.toLocaleDateString("en-CA", { timeZone: "Asia/Kolkata" });
   430→      startDate = dbs + "T00:00:00+05:30";
   431→      endDate = dbs + "T23:59:59.999+05:30";
   432→    } else if (dateRange === "7days") {
   433→      const d7 = new Date(now); d7.setDate(d7.getDate() - 6);
   434→      startDate = d7.toLocaleDateString("en-CA", { timeZone: "Asia/Kolkata" }) + "T00:00:00+05:30";
   435→      endDate = istToday + "T23:59:59.999+05:30";
   436→    }
   437→
   438→    // Build date filter conditions
   439→    const dateFilters = [];
   440→    if (startDate) dateFilters.push(gte(billsTable.createdAt, new Date(startDate)));
   441→    if (endDate) dateFilters.push(lt(billsTable.createdAt, new Date(endDate)));
   442→
   443→    // Bills that have at least one Form-F-required test (distinct on bill)
   444→    const billsWithFormFTests = await db
   445→      .selectDistinct({
   446→        billId: billsTable.id,
   447→        billNumber: billsTable.billNumber,
   448→        patientId: billsTable.patientId,
   449→        orderId: billsTable.orderId,
   450→        createdAt: billsTable.createdAt,
   451→      })
   452→      .from(billsTable)
   453→      .innerJoin(ordersTable, eq(billsTable.orderId, ordersTable.id))
   454→      .innerJoin(orderTestsTable, eq(orderTestsTable.orderId, ordersTable.id))
   455→      .where(and(inArray(orderTestsTable.testId, formFTestIds), ...dateFilters))
   456→      .orderBy(desc(billsTable.createdAt));
   457→
   458→    if (billsWithFormFTests.length === 0) { res.json([]); return; }
   459→
   460→    // Bill IDs that already have Form F records
   461→    const savedRecords = await db
   462→      .select({ billId: formFRecordsTable.billId })
   463→      .from(formFRecordsTable)
   464→      .where(isNotNull(formFRecordsTable.billId));
   465→    const savedBillIdSet = new Set(savedRecords.map((r) => r.billId).filter(Boolean));
   466→
   467→    const pendingBills = billsWithFormFTests.filter((b) => !savedBillIdSet.has(b.billId));
   468→    if (pendingBills.length === 0) { res.json([]); return; }
   469→
   470→    // Patient details
   471→    const patientIds = [...new Set(pendingBills.map((b) => b.patientId).filter(Boolean))] as number[];
   472→    const patients = patientIds.length > 0
   473→      ? await db.select().from(patientsTable).where(inArray(patientsTable.id, patientIds))
   474→      : [];
   475→    const patientMap = new Map(patients.map((p) => [p.id, p]));
   476→
   477→    // Referring doctors per order
   478→    const orderIds = [...new Set(pendingBills.map((b) => b.orderId))] as number[];
   479→    const orders = orderIds.length > 0
   480→      ? await db.select().from(ordersTable).where(inArray(ordersTable.id, orderIds))
   481→      : [];
   482→    const orderMap = new Map(orders.map((o) => [o.id, o]));
   483→
   484→    const doctorIdSet = [...new Set(orders.map((o) => o.doctorId).filter(Boolean))] as number[];
   485→    const doctors = doctorIdSet.length > 0
   486→      ? await db.select().from(doctorsTable).where(inArray(doctorsTable.id, doctorIdSet))
   487→      : [];
   488→    const doctorMap = new Map(doctors.map((d) => [d.id, d]));
   489→
   490→    // Form-F tests per order
   491→    const allOrderTests = orderIds.length > 0
   492→      ? await db
   493→          .select({ orderId: orderTestsTable.orderId, testId: orderTestsTable.testId, testName: testsTable.name })
   494→          .from(orderTestsTable)
   495→          .leftJoin(testsTable, eq(orderTestsTable.testId, testsTable.id))
   496→          .where(inArray(orderTestsTable.orderId, orderIds))
   497→      : [];
   498→
   499→    const orderFormFTestsMap = new Map<number, string[]>();
   500→    for (const ot of allOrderTests) {

   873→formFRouter.post("/latest-scan-proxy", requireStaffPermission("/form-f"), async (req, res) => {
   874→  try {
   875→    const body = req.body ?? {};
   876→    const bridgeUrl = String(body.bridgeUrl ?? SCAN_BRIDGE_URL).trim();
   877→    const mode = String(body.mode ?? "watch").trim(); // "watch" or "direct"
   878→    const useSharp = typeof req.app?.get === "function" ? req.app.get("useSharp") !== false : true;
   879→
   880→    // 1) Fetch from bridge (either /latest-scan or /scan)
   881→    const endpoint = mode === "direct" ? "scan" : "latest-scan";
   882→    const r = await fetch(`${bridgeUrl}/${endpoint}`, { method: "POST", mode: "cors" });
   883→    const j = await r.json().catch(() => ({})) as Record<string, unknown>;
   884→    if (!r.ok || !j.ok) {
   885→      res.status(r.status === 404 ? 404 : 502).json({
   886→        ok: false,
   887→        error: String(j.error || "Bridge failed"),
   888→        code: j.code ? String(j.code) : null,
   889→        fallback: j.fallback ? String(j.fallback) : null,
   890→      });
   891→      return;
   892→    }
   893→
   894→    const imageBase64 = String(j.imageBase64 ?? "");
   895→    const mimeType = String(j.mimeType ?? "image/jpeg");
   896→    const filename = String(j.filename ?? "scan");
   897→    const mtimeMs = Number(j.mtimeMs ?? 0);
   898→    const cacheKey = `${filename}:${mtimeMs}`;
   899→
   900→    // 2) Duplicate protection — reject same file/mtime within last 5 minutes
   901→    const now = Date.now();
   902→    const lastSeen = importedScanCache.get(cacheKey);
   903→    if (lastSeen && now - lastSeen < 5 * 60 * 1000) {
   904→      res.status(409).json({
   905→        ok: false,
   906→        error: "This scan was already imported recently. Wait a few minutes or scan a new document.",
   907→        duplicate: true,
   908→        cacheKey,
   909→      });
   910→      return;
   911→    }
   912→    importedScanCache.set(cacheKey, now);
   913→    // Clean old cache entries (older than 10 min)
   914→    for (const [k, t] of importedScanCache) {
   915→      if (now - t > 10 * 60 * 1000) importedScanCache.delete(k);
   916→    }
   917→
   918→    // 3) Image optimization using Sharp if available
   919→    let optimizedBase64 = imageBase64;
   920→    let optimizedMime = mimeType;
   921→    const maxWidth = Number(body.maxWidth ?? 1200);
   922→    const jpegQuality = Number(body.jpegQuality ?? 85);
   923→    if (useSharp && !mimeType.includes("pdf")) {
   924→      try {
   925→        const sharp = await import("sharp");
   926→        const inputBuf = Buffer.from(imageBase64, "base64");
   927→        let pipeline = sharp.default(inputBuf).rotate();
   928→        if (maxWidth > 0) {
   929→          pipeline = pipeline.resize({ width: maxWidth, withoutEnlargement: true });
   930→        }
   931→        const outBuf = await pipeline.jpeg({ quality: jpegQuality, mozjpeg: true }).toBuffer();
   932→        optimizedBase64 = outBuf.toString("base64");
   933→        optimizedMime = "image/jpeg";
   934→      } catch {
   935→        // Sharp not available or failed — return raw
   936→      }
   937→    }
   938→
   939→    res.json({
   940→      ok: true,
   941→      imageBase64: optimizedBase64,
   942→      mimeType: optimizedMime,
   943→      filename,
   944→      cacheKey,
   945→      optimized: optimizedBase64 !== imageBase64,
   946→    });
   947→  } catch (err) {
   948→    const msg = err instanceof Error ? err.message : "Internal server error";
   949→    req.log?.warn?.({ err }, "latest-scan-proxy error");
